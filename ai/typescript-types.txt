BASIC TYPED USAGE
-----------------
<script lang="ts">
  import { Tree } from '@keenmate/svelte-treeview';
  import type { LTreeNode } from '@keenmate/svelte-treeview';

  interface MyItem {
    id: string;
    path: string;
    name: string;
    type: 'file' | 'folder';
  }

  let data: MyItem[] = $state([]);
  let selectedNode: LTreeNode<MyItem> | null = $state(null);

<Tree
  {data}
  idMember="id"
  pathMember="path"
  bind:selectedNode
/>

LTREENODE INTERFACE
-------------------
interface LTreeNode<T> {
  // Path information
  path: string;              // Full path "1.2.3"
  pathSegment: string;       // Last segment "3"
  parentPath: string | null; // Parent "1.2" or null

  // Structure
  level: number | null;      // Depth (0 = root)
  children: Record<string, LTreeNode<T>>;
  hasChildren: boolean;

  // Your data
  data: T | null;            // Original data object

  // State
  isExpanded: boolean;
  isSelected: boolean;

  // Drag and drop
  isDraggable: boolean;
  isDropAllowed: boolean;
  allowedDropPositions: DropPosition[] | null | undefined;
}

DROPPOSITION TYPE
-----------------
type DropPosition = 'before' | 'after' | 'child';

Usage:
import type { DropPosition } from '@keenmate/svelte-treeview';

function getAllowedPositions(node: LTreeNode<MyItem>): DropPosition[] | null {
  if (node.data?.type === 'file') {
    return ['before', 'after'];
  }
  return null;  // all positions
}

CONTEXT MENU TYPES (v5.0.0-rc03+)
----------------------------------
interface ContextMenuItem {
  id?: string;
  label: string;              // was "title" in old API
  icon?: string;
  shortcut?: string;          // e.g. "Ctrl+C", "F2"
  isDisabled?: boolean;
  isVisible?: boolean;
  className?: string;         // e.g. "danger"
  onclick?: () => void | Promise<void>;  // was "callback" in old API
  children?: ContextMenuEntry[];  // submenu items
}

interface ContextMenuDivider {
  divider: true;              // was "isDivider" flag in old API
  label?: string;             // named divider: ──── [label] ────
}

type ContextMenuEntry = ContextMenuItem | ContextMenuDivider;

Usage:
function createMenu(
  node: LTreeNode<MyItem>,
  closeMenu: () => void,
  selectedNodes?: LTreeNode<MyItem>[]
): ContextMenuEntry[] {
  return [
    { label: 'Open', icon: '📂', shortcut: 'Enter', onclick: () => openItem(node.data) },
    { divider: true, label: 'Edit' },
    { label: 'Delete', icon: '🗑️', className: 'danger', onclick: () => deleteItem(node.data) }
  ];
}

INSERTARRAYRESULT INTERFACE
---------------------------
interface InsertArrayResult<T> {
  successful: number;
  failed: Array<{
    node: LTreeNode<T>;
    originalData: T;
    error: string;
  }>;
  total: number;
}

Usage:
let insertResult: InsertArrayResult<MyItem> | undefined = $state();

$effect(() => {
  if (insertResult) {
    console.log(`${insertResult.successful} succeeded`);
    console.log(`${insertResult.failed.length} failed`);
  }
});

COMPLEX DATA TYPES
------------------
// Union types
interface FileItem {
  id: string;
  path: string;
  name: string;
  type: 'file';
  size: number;
  extension: string;
}

interface FolderItem {
  id: string;
  path: string;
  name: string;
  type: 'folder';
  childCount: number;
}

type TreeItem = FileItem | FolderItem;

// Type guards
function isFile(item: TreeItem): item is FileItem {
  return item.type === 'file';
}

function isFolder(item: TreeItem): item is FolderItem {
  return item.type === 'folder';
}

// Usage in callbacks
<Tree
  data={items as TreeItem[]}
  getDisplayValueCallback={(node) => {
    if (!node.data) return '';
    if (isFile(node.data)) {
      return `${node.data.name} (${node.data.size}KB)`;
    }
    return `${node.data.name} (${node.data.childCount} items)`;
  }}
/>

STRICT NULL CHECKS
------------------
Handle nullable data:

onNodeClicked={(node) => {
  // node.data can be null
  if (node.data) {
    console.log(node.data.name);
  }
}}

// Or with optional chaining
onNodeClicked={(node) => {
  console.log(node.data?.name ?? 'Unknown');
}}

TYPE NARROWING IN TEMPLATES
---------------------------
<Tree {data}>
  {#snippet nodeTemplate(node)}
    {#if node.data}
      <span>{node.data.name}</span>
      {#if isFolder(node.data)}
        <span class="count">({node.data.childCount})</span>
      {/if}
    {/if}
  {/snippet}
</Tree>

EXTENDING TYPES
---------------
// Extend base node data
interface BaseItem {
  id: string;
  path: string;
  name: string;
}

interface ExtendedItem extends BaseItem {
  createdAt: Date;
  updatedAt: Date;
  metadata: Record<string, unknown>;
}

// Use extended type
let data: ExtendedItem[] = $state.raw([]);

<Tree
  {data}
  idMember="id"
  pathMember="path"
  getDisplayValueCallback={(node) => {
    const item = node.data;
    if (!item) return '';
    return `${item.name} (${item.createdAt.toLocaleDateString()})`;
  }}
/>

BEST PRACTICES
--------------
✅ Define interface for your data type
✅ Use type imports (import type { ... })
✅ Handle null/undefined in callbacks
✅ Use type guards for union types
✅ Type tree reference with generic

❌ Don't use `any` type
❌ Don't ignore null checks on node.data
❌ Don't cast without type guards
❌ Don't forget generic parameter on Tree ref
