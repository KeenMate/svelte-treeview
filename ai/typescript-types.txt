LTREENODE INTERFACE
-------------------
interface LTreeNode<T> {
  // Path information
  path: string;              // Full path "1.2.3"
  pathSegment: string;       // Last segment "3"
  parentPath: string | null; // Parent "1.2" or null

  // Structure
  level: number | null;      // Depth (0 = root)
  children: Record<string, LTreeNode<T>>;
  hasChildren: boolean;

  // Your data
  data: T | null;            // Original data object

  // State
  isExpanded: boolean;
  isSelected: boolean;

  // Drag and drop
  isDraggable: boolean;
  isDropAllowed: boolean;
  allowedDropPositions: DropPosition[] | null | undefined;
}

DROPPOSITION TYPE
-----------------
type DropPosition = 'before' | 'after' | 'child';

Usage:
import type { DropPosition } from '@keenmate/svelte-treeview';

function getAllowedPositions(node: LTreeNode<MyItem>): DropPosition[] | null {
  if (node.data?.type === 'file') {
    return ['before', 'after'];
  }
  return null;  // all positions
}

CONTEXT MENU TYPES
------------------
interface ContextMenuItem {
  id?: string;
  label: string;              // was "title" in old API
  icon?: string;
  shortcut?: string;          // e.g. "Ctrl+C", "F2"
  isDisabled?: boolean;
  isVisible?: boolean;
  className?: string;         // e.g. "danger"
  onclick?: () => void | Promise<void>;  // was "callback" in old API
  shouldCloseOnClick?: boolean;   // default true; false keeps menu open (handler closes via close())
  children?: ContextMenuEntry[];  // submenu items
}

interface ContextMenuDivider {
  divider: true;              // was "isDivider" flag in old API
  label?: string;             // named divider: ──── [label] ────
}

type ContextMenuEntry = ContextMenuItem | ContextMenuDivider;

Usage:
function createMenu(
  node: LTreeNode<MyItem>,
  close: () => void
): ContextMenuEntry[] {
  return [
    { label: 'Open', icon: '📂', shortcut: 'Enter', onclick: () => openItem(node.data) },
    { divider: true, label: 'Edit' },
    { label: 'Delete', icon: '🗑️', className: 'danger', onclick: () => deleteItem(node.data) }
  ];
}

EVENT TYPES (on* — fire-and-forget, ONE context object)
-------------------------------------------------------
// All on* events take a single context object, NOT positional args.
// on* = notification (return value ignored, except onTreeKeydown which intercepts).

onNodeClick?: (ctx: NodeEventContext<T>) => void;        // ctx.node = clicked node
onNodeDoubleClick?: (ctx: NodeEventContext<T>) => void;
onNodeDragStart?: (ctx: NodeDragContext<T>) => void;     // grabbed node + event + dragged set
onNodeDragOver?: (ctx: NodeDragContext<T>) => void;      // hovered node + event + dragged set
onNodeDrop?: (ctx: NodeDropContext<T>) => void;          // source/target/dragged/dropped/...
onHighlightChange?: (ctx: SelectionChangeContext<T>) => void;  // UI multi-select (highlightedPaths)
onSelectionChange?: (ctx: SelectionChangeContext<T>) => void;  // checkbox set (selectedPaths)
onCopy?: (ctx: ClipboardEventContext<T>) => void;
onCut?: (ctx: ClipboardEventContext<T>) => void;
onDelete?: (ctx: ClipboardEventContext<T>) => void;      // ctx.operation omitted for delete
onPaste?: (result: PasteResult<T>) => void;

// Interceptor: return true to suppress default + built-in shortcuts.
onTreeKeydown?: (ctx: {
  event: KeyboardEvent;
  focusedNode: LTreeNode<T> | null;
  highlightedNodes: LTreeNode<T>[];
  controller: TreeController<T>;
}) => boolean | void;

CALLBACK TYPES (before*/get*/*TransformationCallback — intercept / provide data)
-------------------------------------------------------------------------------
// beforeDropCallback is DELIBERATELY still 5-arg positional (drop pair is asymmetric).
beforeDropCallback?: (
  dropNode: LTreeNode<T> | null,
  draggedNode: LTreeNode<T>,
  position: DropPosition,
  event: DragEvent,
  operation: DropOperation
) => boolean | { position?: DropPosition; operation?: DropOperation } | void | Promise<...>;

// Clipboard batch-policy interceptors (single context object).
// Return a new path[] to rewrite the set, false to block, or nothing to proceed.
beforeCopyCallback?: (ctx: BeforeCopyContext<T>) => string[] | false | void;
beforeCutCallback?: (ctx: BeforeCopyContext<T>) => string[] | false | void;
beforeDeleteCallback?: (ctx: BeforeDeleteContext<T>) => string[] | false | void;
// Return { targetPath?, position? } to redirect, false to block, or nothing.
beforePasteCallback?: (ctx: BeforePasteContext<T>) => { targetPath?: string; position?: DropPosition } | false | void;

// Per-node transforms — operation-neutral, by DIRECTION. Both fire for clipboard AND drag.
nodeOutputTransformationCallback?: (data: T, ctx: NodeTransformContext<T>) => T;         // egress: node leaves source (phase 'output')
nodeInputTransformationCallback?: (data: T, ctx: NodeTransformContext<T>) => T | null;   // ingress: duplicate lands (phase 'input'); null skips node/subtree

// Value getters
getDisplayValueCallback?: (node: LTreeNode<T>) => string;
getSearchValueCallback?: (node: LTreeNode<T>) => string;
getAllowedDropPositionsCallback?: (node: LTreeNode<T>) => DropPosition[] | null | undefined;

// Context menu (label/onclick/divider API). closeMenuCallback closes the menu;
// selectedNodes is the current highlight set when the menu opened.
getContextMenuItemsCallback?: (
  node: LTreeNode<T>,
  closeMenuCallback: () => void,
  selectedNodes?: LTreeNode<T>[]
) => ContextMenuEntry[];

// Sort
sortCallback?: (items: LTreeNode<T>[]) => LTreeNode<T>[];

CONTEXT TYPES (shared shapes — exported for consumers)
------------------------------------------------------
// A pointer to one node plus the relational context the tree already knows.
// node/parent/siblings are LIVE when reachable, null/[] cross-tree or after a cut.
interface NodeRef<T> {
  path: string;
  node: LTreeNode<T> | null;
  parent: LTreeNode<T> | null;
  siblings: LTreeNode<T>[];  // children of parent (includes node); top-level nodes when parent is null
}

// onNodeClick / onNodeDoubleClick receive a bare NodeRef.
type NodeEventContext<T> = NodeRef<T>;

// onNodeDragStart / onNodeDragOver. A drag is single-origin → fires ONCE even
// for a multi-selection; `dragged` carries the full top-level set.
interface NodeDragContext<T> extends NodeRef<T> {
  event: DragEvent;
  dragged: NodeRef<T>[];  // full top-level set (single-item array for a single drag)
}

// onNodeDrop. `dropped` = nodes the library placed (move: moved; copy: fresh copies);
// null cross-tree or when shouldAutoHandleMove/Copy=false (consumer inserts).
interface NodeDropContext<T> {
  source: NodeRef<T>;                    // lead dragged node
  target: NodeRef<T> | null;            // drop node; null = empty / root
  dragged: NodeRef<T>[];                // full top-level dragged set (includes source)
  dropped: NodeRef<T>[] | null;         // placed nodes; null when library didn't place them
  position: DropPosition;
  operation: DropOperation;
  event: DragEvent | TouchEvent;
}

// onCopy / onCut / onDelete. operation omitted for delete. For cut/delete,
// nodes are PRE-REMOVAL snapshots (the tree no longer holds them at fire time).
interface ClipboardEventContext<T> {
  operation?: 'copy' | 'cut';
  paths: string[];
  nodes: LTreeNode<T>[];
}

// onHighlightChange (highlightedPaths) / onSelectionChange (selectedPaths).
// Suppressed by { silent: true } mutations.
interface SelectionChangeContext<T> {
  paths: Set<string>;
  nodes: LTreeNode<T>[];
}

TRANSFORM / CLIPBOARD CONTEXT TYPES
-----------------------------------
// Passed to BOTH output & input transforms, one per node. Symmetric source/target:
// same field, same meaning. target + position are null during the output phase.
interface NodeTransformContext<T> {
  operation: 'copy' | 'cut';
  phase: 'output' | 'input';                     // egress (node leaves) vs. ingress (duplicate lands)
  isRoot: boolean;                               // top-level node vs. descendant riding along
  index: number;                                 // index of the root entry in this batch
  position: 'child' | 'before' | 'after' | null; // how roots land relative to target.node
  source: NodeRef<T>;                            // origin side (live same-tree, null/[] cross-tree/cut)
  target: NodeRef<T> | null;                     // destination side (null during output phase)
}

interface BeforeCopyContext<T> {
  operation: 'copy' | 'cut';
  paths: string[];
  nodes: LTreeNode<T>[];  // resolved live nodes (missing paths dropped)
}

interface BeforeDeleteContext<T> {
  paths: string[];        // top-level paths about to be deleted
  nodes: LTreeNode<T>[];
}

interface BeforePasteContext<T> {
  operation: 'copy' | 'cut';
  target: { path: string; node: LTreeNode<T> | null };  // path '' = root; may be redirected
  entries: readonly ClipboardEntry<T>[];                // immutable clipboard snapshots (not live nodes)
}

COMPLEX DATA TYPES
------------------
// Union types
interface FileItem {
  id: string;
  path: string;
  name: string;
  type: 'file';
  size: number;
  extension: string;
}

interface FolderItem {
  id: string;
  path: string;
  name: string;
  type: 'folder';
  childCount: number;
}

type TreeItem = FileItem | FolderItem;

// Type guards
function isFile(item: TreeItem): item is FileItem {
  return item.type === 'file';
}

function isFolder(item: TreeItem): item is FolderItem {
  return item.type === 'folder';
}

// Usage in callbacks
<Tree
  data={items as TreeItem[]}
  getDisplayValueCallback={(node) => {
    if (!node.data) return '';
    if (isFile(node.data)) {
      return `${node.data.name} (${node.data.size}KB)`;
    }
    return `${node.data.name} (${node.data.childCount} items)`;
  }}
/>

STRICT NULL CHECKS
------------------
Handle nullable data (ctx.node can be null cross-tree/after-cut; ctx.node.data can be null):

onNodeClick={(ctx) => {
  // ctx.node and ctx.node.data can both be null
  if (ctx.node?.data) {
    console.log(ctx.node.data.name);
  }
}}

// Or with optional chaining
onNodeClick={(ctx) => {
  console.log(ctx.node?.data?.name ?? 'Unknown');
}}

TYPE NARROWING IN TEMPLATES
---------------------------

    {/if}
  {/snippet}
</Tree>

BEST PRACTICES
--------------
✅ Define interface for your data type
✅ Use type imports (import type { ... })
✅ Handle null/undefined in callbacks
✅ Use type guards for union types
✅ Type tree reference with generic

❌ Don't use `any` type
❌ Don't ignore null checks on node.data
❌ Don't cast without type guards
❌ Don't forget generic parameter on Tree ref
